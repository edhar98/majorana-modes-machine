\begin{frame}{The Project in One Slide}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \textbf{The bridge: fermions $\to$ qubits $\to$ NISQ}
      \begin{itemize}
        \item The \textbf{1D Kitaev chain} is the minimal model of a 1D topological superconductor: in the topological phase it hosts \textbf{Majorana zero modes} localized at its two ends.
        \item A \textbf{Jordan--Wigner} map turns that fermionic model into an exact qubit Hamiltonian --- so a quantum device can, in principle, host the physics.
        \item We prepare its ground state by \textbf{VQE} and read the topological signature from a circuit-measurable observable.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \small
      \begin{block}{The question}
        Can a \emph{topological signature} be prepared and still \emph{survive} on a noisy NISQ device --- and how far in system size $L$ can it be pushed?
      \end{block}
      \vspace{0.15cm}
      \footnotesize
      \textbf{Four Blocks:}
      \begin{enumerate}
        \item the physics bridge,
        \item finite-size physics \& qubit encoding,
        \item measuring topology with circuits,
        \item the NISQ reality check (results).
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Block 1 --- The Physics Bridge}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.72\textheight,keepaspectratio]{block1_03_phase_diagram.pdf}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      The Kitaev chain: $p$-wave pairing on a 1D lattice,
      \[
        E_k=\sqrt{(\mu+2t\cos k)^2+(2\Delta\sin k)^2}.
      \]
      \begin{itemize}
        \item \textbf{Bulk gap} closes only at $|\mu|=2t$ --- the phase boundary.
        \item \textbf{Topological phase} for $|\mu|<2t$: a $\mathbb{Z}_2$ \textbf{winding number} $\nu=1$; trivial ($\nu=0$) outside.
        \item Bulk--boundary correspondence $\Rightarrow$ \textbf{Majorana zero modes} pinned to the ends in the topological phase.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ============================ BLOCK 2 =====================================
\begin{frame}{Block 2 --- Finite-Size Physics \& Qubit Encoding}
  \centering
  \includegraphics[width=\textwidth,height=0.46\textheight,keepaspectratio]{block2_01_parity_gap_vs_bdg.pdf}
  \par\vspace{0.08cm}
  {\footnotesize\raggedright
  On a finite chain the two end Majoranas overlap and split, $\Delta_P(L)\sim e^{-L/\xi}$.
  \begin{itemize}\itemsep2pt
    \item \textbf{Exponential protection}: the parity gap shrinks as the chain grows --- longer is better.
    \item \textbf{Jordan--Wigner} gives the \emph{exact} qubit Hamiltonian
      $H=-\tfrac{\mu}{2}\sum_j Z_j+\tfrac{t-\Delta}{2}\sum_j X_jX_{j+1}+\tfrac{t+\Delta}{2}\sum_j Y_jY_{j+1}$.
    \item \textbf{Parity} $\Pop=\prod_j Z_j$ is a $\mathbb{Z}_2$ symmetry: even/odd sectors label the two near-degenerate ground states.
  \end{itemize}\par}
\end{frame}

\begin{frame}{Block 4 --- NISQ Reality Check III: Scaling to Large $L$ (Methods)}
  \footnotesize
  Past the dense $L\!\approx\!15$ wall: remove every exponential object but the mild $2^L$ statevector.
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Removing the two exponentials}
        \begin{itemize}\itemsep2pt
          \item \textbf{ED $\to$ free fermions}: the model is quadratic --- exact GS \& edge string from a $2L\times2L$ \textbf{Bogoliubov--de~Gennes} matrix, $O(L^3)$.
          \item \textbf{$4^L$ density matrix $\to$ MPS trajectories}: each noisy shot is one \textbf{matrix-product state (MPS)} trajectory --- a chain of per-site $\chi\!\times\!\chi$ tensors with bond dimension $\chi\le2^{r}$ ($=16$ at $r^\ast{=}4$), so memory is $O(L\chi^2)$: \textbf{linear in $L$}, not $4^L$.
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{block}{The mild object + HPC}
        \begin{itemize}\itemsep2pt
          \item \textbf{Ideal VQE}: a $2^L$ statevector with \emph{sparse} Pauli operators --- never the $4^L$ dense $H$ ($\sim\!500\times$ faster at $L=14$).
          \item $\mu=0.5t$ (gapped), L-BFGS-B, ED-free even-sector selection.
          \item $(L,\text{reps})$ grid parallelized on the \textbf{Leipzig SC cluster}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Block 4 --- Which Physical Errors Kill the Topology?}
  \small
  A perturbation is harmless only if it is \textbf{(i) local}, \textbf{(ii) fermion-parity preserving}, and \textbf{(iii) weaker than the bulk gap $\Delta$}. Break any one and the phase dies.
  \vspace{0.15cm}
  \begin{center}
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Physical error (Kitaev $H$)} & \textbf{Jordan--Wigner $\to$ qubit} & \textbf{Verdict} \\
    \midrule
    Charge noise $\delta\mu_i\,c_i^\dagger c_i$ & $\sum_i\delta h_i\,Z_i$ \; (parity-even) & \textcolor{topological}{\textbf{robust}} (up to $\Delta$) \\
    Quasiparticle poisoning $c_i^\dagger$ & $\bigl(\prod_{k<i}Z_k\bigr)(X_i\!\mp\!iY_i)$ \; (parity-\emph{odd}) & \textcolor{trivial}{\textbf{fatal}}: flips the bit \\
    Gap closure $\mu\!\to\!2t$ & $\sum_i Z_i$ dominates $\sum_i X_iX_{i+1}$ & \textcolor{trivial}{\textbf{fatal}}: phase transition \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.1cm}
  \footnotesize
  \textbf{What to measure:} the string order $O_{\mathrm{edge}}$, the fermion parity $P=\prod_j Z_j$, and the splitting $\delta E=|E_{\mathrm{even}}-E_{\mathrm{odd}}|\sim e^{-L/\xi}$.
\end{frame}

\begin{frame}{Block 4 --- Robust Error: Local Disorder (Exact, to $L=32$)}
  \centering
  \includegraphics[width=\textwidth,height=0.52\textheight,keepaspectratio]{block4_disorder_robustness.pdf}
  \par\vspace{0.1cm}
  {\footnotesize\raggedright
  Charge noise $=$ a random field $\sum_j\delta h_j Z_j$ (parity-even): per-site $\mu_j=\mu_0+\mathrm{Uniform}(-W,W)$, where $W$ is the disorder half-width (the $x$-axis). Free-fermion BdG, exact.
  \begin{itemize}\itemsep2pt
    \item Below the bulk gap $\Delta$ the edge string \textbf{holds} and the splitting stays \emph{exponentially} small.
    \item Past $W\!\sim\!\Delta$ the gap closes and both die.
  \end{itemize}
  \textbf{Local, parity-even, sub-gap $\Rightarrow$ topologically harmless.}\par}
\end{frame}

\begin{frame}{Block 4 --- Fatal Error: Quasiparticle Poisoning}

  \includegraphics[width=0.92\textwidth,height=0.48\textheight,keepaspectratio]{block4_robust_vs_fatal.pdf}
  \par\vspace{0.1cm}

  A stray electron $c_0+c_0^\dagger=\gamma_0=X_0$ is \textbf{parity-odd}. Exact diagonalisation, $L=10$.
  \begin{itemize}\itemsep2pt
    \item \textcolor{topological}{Parity-even} $Z$-disorder: $\langle P\rangle{=}{+}1$, edge string survives.

  \textbf{QP poisoning is the Achilles heel}: the poisoning time must beat the gate time.\par}
\end{frame}

% ============================ CLOSING =====================================
\begin{frame}{Takeaways}
  \begin{itemize}
    \item The Kitaev chain's Majorana edge signature can be \textbf{prepared and measured on qubits} via a non-local string order $O_{\mathrm{edge}}=\Oedge$.
    \item \textbf{Parity (symmetry) is protected} under noise; the \textbf{edge string (topology) is not} noise-immune at fixed $L$.
    \item Classically, both exponentials are removable: the ED reference $\to$ \textbf{free fermions} (BdG), the $4^L$ density matrix $\to$ \textbf{MPS trajectories} --- only a mild $2^L$ statevector remains.
    \item For the edge-string diagnostic, \textbf{expressibility is $L$-independent} (constant depth $r^\ast=4$): the large-$L$ wall is \emph{decoherence} over a linearly growing gate count.
    \item On \textbf{real IBM noise} the signature survives far beyond the toy model ($p=0.05$), staying measurable to $L\!\sim\!48$ across devices.
    \item \textbf{Which errors kill topology?} Local parity-even sub-gap noise is harmless; \textbf{quasiparticle poisoning} (parity-odd) and gap closure are fatal.
  \end{itemize}
